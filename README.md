// ===== CALENDAR =====
function renderCalendar() {
  const u = getUser();
  const ut = state.role==='admin' ? tasks : getUserTasks(u.id);
  const cd = state.calendarDate;
  const view = state.calendarView;
  const year = cd.getFullYear(); const month = cd.getMonth();
  const monthNames = ['January','February','March','April','May','June','July','August','September','October','November','December'];

  let content = '';
  if(view==='month') {
    const firstDay = new Date(year,month,1).getDay(); // 0=Sun
    const daysInMonth = new Date(year,month+1,0).getDate();
    const dayNames = ['Sun','Mon','Tue','Wed','Thu','Fri','Sat'];
    let cells = '';
    for(let i=0;i<firstDay;i++) cells += '<div class="p-1"></div>';
    for(let d=1;d<=daysInMonth;d++){
      const dateStr = `${year}-${String(month+1).padStart(2,'0')}-${String(d).padStart(2,'0')}`;
      const dayTasks = ut.filter(t=>t.dueDate===dateStr);
      const isToday = dateStr === '2026-09-08';
      cells += `<div class="p-1 border border-gray-100 rounded-lg min-h-[70px] ${isToday?'bg-blue-50':''}">
        <div class="text-xs font-medium ${isToday?'text-blue-600':'text-gray-600'} mb-1">${d}</div>
        ${dayTasks.slice(0,2).map(t=>`<div class="text-[11px] px-1 py-0.5 rounded mb-0.5 cursor-pointer truncate ${t.status==='Completed'?'bg-green-50 text-green-700':'classifyTask(t)==="Overdue"?'bg-red-50 text-red-700':'bg-blue-50 text-blue-700'}" data-action="view-task" data-id="${t.id}">${esc(t.name)}</div>`).join('')}
        ${dayTasks.length>2?`<div class="text-[10px] text-gray-400">+${dayTasks.length-2} more</div>`:''}
      </div>`;
    }
    content = `<div class="grid grid-cols-7 gap-0.5 text-center mb-1">${dayNames.map(d=>`<div class="text-xs font-semibold text-gray-500 py-1">${d}</div>`).join('')}</div>
      <div class="grid grid-cols-7 gap-0.5">${cells}</div>`;
  } else if(view==='week') {
    const startOfWeek = new Date(cd);
    startOfWeek.setDate(cd.getDate() - cd.getDay());
    let rows = '';
    for(let i=0;i<7;i++){
      const d = new Date(startOfWeek); d.setDate(startOfWeek.getDate()+i);
      const dateStr = d.toISOString().slice(0,10);
      const dayTasks = ut.filter(t=>t.dueDate===dateStr).sort((a,b)=>a.dueTime.localeCompare(b.dueTime));
      const isToday = dateStr === '2026-09-08';
      rows += `<div class="flex gap-3 py-2 border-b border-gray-100 ${isToday?'bg-blue-50 -mx-3 px-3 rounded-lg':''}">
        <div class="w-16 text-sm font-medium ${isToday?'text-blue-600':''}">${['Sun','Mon','Tue','Wed','Thu','Fri','Sat'][d.getDay()]}<br><span class="text-xs text-gray-500">${d.getDate()}</span></div>
        <div class="flex-1 space-y-1">${dayTasks.map(t=>`<div class="text-sm flex items-center gap-2 cursor-pointer hover:bg-gray-50 rounded px-2 py-1" data-action="view-task" data-id="${t.id}">
          <span class="text-xs text-gray-500">${fmtTime(t.dueTime)}</span><span class="font-medium">${esc(t.name)}</span><span class="badge ${statusBadge(t.status)} text-[10px]">${t.status}</span>
        </div>`).join('')||'<div class="text-xs text-gray-400 py-1">No tasks</div>'}</div>
      </div>`;
    }
    content = rows;
  } else {
    const sorted = [...ut].sort((a,b)=>a.dueDate.localeCompare(b.dueDate)||a.dueTime.localeCompare(b.dueTime));
    content = sorted.length ? sorted.map(t=>`<div class="flex items-center justify-between p-3 border-b border-gray-100 cursor-pointer hover:bg-gray-50" data-action="view-task" data-id="${t.id}">
      <div><div class="font-medium text-sm">${esc(t.name)}</div><div class="text-xs text-gray-500">${fmt(t.dueDate)}, ${fmtTime(t.dueTime)}</div></div>
      <div class="flex gap-1"><span class="badge ${priorityBadge(t.priority)} text-[10px]">${t.priority}</span><span class="badge ${statusBadge(t.status)} text-[10px]">${t.status}</span></div>
    </div>`).join('') : '<div class="text-center text-gray-400 py-8">No tasks</div>';
  }

  return `<div class="max-w-5xl mx-auto space-y-4">
    <div class="flex flex-wrap items-center justify-between gap-3">
      <h1 class="text-xl font-bold">Calendar</h1>
      <div class="flex gap-1">${['month','week','list'].map(v=>`<button class="tab ${state.calendarView===v?'active':''}" data-action="cal-view" data-view="${v}">${v[0].toUpperCase()+v.slice(1)}</button>`).join('')}</div>
    </div>
    <div class="card">
      <div class="flex items-center justify-between mb-4">
        <button class="btn btn-secondary p-2" data-action="cal-prev" aria-label="Previous"><i data-lucide="chevron-left" style="width:16px;height:16px"></i></button>
        <span class="font-semibold">${monthNames[month]} ${year}</span>
        <button class="btn btn-secondary p-2" data-action="cal-next" aria-label="Next"><i data-lucide="chevron-right" style="width:16px;height:16px"></i></button>
      </div>
      ${content}
    </div>
  </div>`;
}

// ===== EVENTS =====
function bindEvents() {
  // Login
  document.getElementById('login-btn')?.addEventListener('click', doLogin);

    document.getElementById('staff-pin').value='1234';
    doLogin();
  });
  document.getElementById('staff-pin')?.addEventListener('keydown', e=>{ if(e.key==='Enter') doLogin(); });

  // Nav
  document.querySelectorAll('[data-nav]').forEach(el=>el.addEventListener('click', ()=>{
    const nav = el.dataset.nav;
    if(nav==='logout') { state.loggedIn=false; state.page='login'; state.currentUser=null; state.editingProfile=false; toast('Logged out successfully'); render(); return; }

    if(window.innerWidth<=768) state.sidebarOpen=false;
    render();
  }));

  document.getElementById('toggle-sidebar')?.addEventListener('click', ()=>{ state.sidebarOpen=!state.sidebarOpen; render(); });
  document.getElementById('notif-btn')?.addEventListener('click', e=>{ e.stopPropagation(); state.showNotifDropdown=!state.showNotifDropdown; render(); });

  document.getElementById('task-search')?.addEventListener('input', e=>{ state.taskFilter.search=e.target.value; render(); });
  document.getElementById('filter-status')?.addEventListener('change', e=>{ state.taskFilter.status=e.target.value; render(); });
  document.getElementById('filter-priority')?.addEventListener('change', e=>{ state.taskFilter.priority=e.target.value; render(); });
  document.getElementById('filter-category')?.addEventListener('change', e=>{ state.taskFilter.category=e.target.value; render(); });
  document.getElementById('log-date')?.addEventListener('change', e=>{ state.workLogDateFilter=e.target.value; render(); });

  // Settings

function doLogin() {
  const id = document.getElementById('staff-id')?.value.trim().toUpperCase();
  const pin = document.getElementById('staff-pin')?.value.trim();
  const errEl = document.getElementById('login-error');
  if(!id||!pin) { errEl.textContent='Please enter both Staff ID and PIN.'; errEl.classList.remove('hidden'); return; }
  if(id==='NC001'&&pin==='1234') {
    state.loggedIn=true; state.page='dashboard'; state.currentUser=staffMembers[0];
    toast('Welcome back, '+staffMembers[0].name);
    render();
  } else {
    errEl.textContent='Incorrect Staff ID or PIN. Try demo: NC001 / 1234'; errEl.classList.remove('hidden');
  }
}

function handleAction(e) {
  const action = e.currentTarget.dataset.action;
  const id = e.currentTarget.dataset.id;

  if(action==='add-task') { state.modal={type:'task-form',data:{assignee:getUser().id}}; renderModalOverlay(); }
  else if(action==='edit-task') { const t=tasks.find(t=>t.id==id); if(t){state.modal={type:'task-form',data:{...t}}; renderModalOverlay();} }
  else if(action==='delete-task') { state.modal={type:'confirm-delete',data:{deleteType:'task',id:parseInt(id)}}; renderModalOverlay(); }
  else if(action==='complete-task') { const t=tasks.find(t=>t.id==id); if(t){t.status='Completed'; persist(); toast(t.name+' marked as completed'); render();} }
  else if(action==='view-task') { const t=tasks.find(t=>t.id==id); if(t){state.modal={type:'task-detail',data:t}; renderModalOverlay();} }
  else if(action==='add-log') { state.modal={type:'log-form',data:{assignee:getUser().id}}; renderModalOverlay(); }
  else if(action==='edit-log') { const l=workLogs.find(l=>l.id==id); if(l){state.modal={type:'log-form',data:{...l}}; renderModalOverlay();} }
  else if(action==='delete-log') { state.modal={type:'confirm-delete',data:{deleteType:'log',id:parseInt(id)}}; renderModalOverlay(); }
  else if(action==='close-modal'||action==='close-modal-bg') { state.modal=null; document.getElementById('modal-root')?.remove(); }
  else if(action==='confirm-delete') {
    const d=state.modal.data;
    if(d.deleteType==='task') { tasks=tasks.filter(t=>t.id!==d.id); toast('Task deleted'); }
    else if(d.deleteType==='log') { workLogs=workLogs.filter(l=>l.id!==d.id); toast('Log entry deleted'); }
    state.modal=null; persist(); render();
  }
  else if(action==='edit-profile') { state.editingProfile=true; render(); }
  else if(action==='cancel-profile') { state.editingProfile=false; render(); }
  else if(action==='save-profile') { handleProfileSave(); }
  else if(action==='clear-filters') { state.taskFilter={status:'',priority:'',category:'',search:''}; render(); }
  else if(action==='cal-view') { state.calendarView=e.currentTarget.dataset.view; render(); }
  else if(action==='cal-prev') { state.calendarDate.setMonth(state.calendarDate.getMonth()-1); render(); }
  else if(action==='cal-next') { state.calendarDate.setMonth(state.calendarDate.getMonth()+1); render(); }
  else if(action==='view-member') { state.modal={type:'member-detail',data:id}; renderModalOverlay(); }
  else if(action==='manage-staff') { state.modal={type:'manage-staff'}; renderModalOverlay(); }
}

function render() {
    document.documentElement.classList.toggle('dark', state.darkMode);
    const app = document.getElementById('app');

    if (state.page === 'login') {
        app.innerHTML = renderLogin();
    } else {
        app.innerHTML = renderAppShell();
    }

    if (window.lucide) {
        lucide.createIcons();
    }

    bindEvents();
}
  // Rebind modal events
  root.querySelectorAll('[data-action]').forEach(el=>el.addEventListener('click', handleAction));
  root.querySelector('#task-form')?.addEventListener('submit', handleTaskSubmit);
  root.querySelector('#log-form')?.addEventListener('submit', handleLogSubmit);
}

function handleTaskSubmit(e) {
  e.preventDefault();
  const name = document.getElementById('tf-name').value.trim();
  const idEl = document.getElementById('tf-id');
  const data = {
    name,
    description: document.getElementById('tf-desc').value.trim(),
    category: document.getElementById('tf-cat').value.trim() || 'General',
    priority: document.getElementById('tf-pri').value,
    status: document.getElementById('tf-status').value,
    dueDate: document.getElementById('tf-date').value,
    dueTime: document.getElementById('tf-time').value,
    notes: document.getElementById('tf-notes').value.trim(),
    assignee: getUser().id,
  };
  if(!name||!data.dueDate||!data.dueTime) return;

  if(idEl) {
    const t = tasks.find(t=>t.id==idEl.value);
    if(t) Object.assign(t, data);
    toast('Task updated');
  } else {
    data.id = nextTaskId++;
    tasks.push(data);
    toast('Task created');
  }
  state.modal=null; persist(); render();
}

function handleLogSubmit(e) {
  e.preventDefault();
  const idEl = document.getElementById('lf-id');
  const data = {
    date: document.getElementById('lf-date').value,
    task: document.getElementById('lf-task').value.trim(),
    description: document.getElementById('lf-desc').value.trim(),
    startTime: document.getElementById('lf-start').value,
    endTime: document.getElementById('lf-end').value,
    status: document.getElementById('lf-status').value,
    remarks: document.getElementById('lf-remarks').value.trim(),
    assignee: getUser().id,
  };
  if(!data.task||!data.date||!data.startTime||!data.endTime) return;

  if(idEl) {
    const l = workLogs.find(l=>l.id==idEl.value);
    if(l) Object.assign(l, data);
    toast('Log entry updated');
  } else {
    data.id = nextLogId++;
    workLogs.push(data);
    toast('Work log entry added');
  }
  state.modal=null; persist(); render();
}

function handleProfileSave() {
  const u = getUser();
  const fields = document.querySelectorAll('[data-field]');
  if(!profileEdits[u.id]) profileEdits[u.id] = {};
  fields.forEach(f=>{ profileEdits[u.id][f.dataset.field] = f.value.trim(); });
  state.editingProfile = false;
  persist();
  toast('Profile updated');
  render();
}

function handleAvatarChange(e) {
  const file = e.target.files[0];
  if(!file) return;
  const reader = new FileReader();
  reader.onload = function(ev) {
    const u = getUser();
    if(!profileEdits[u.id]) profileEdits[u.id] = {};
    profileEdits[u.id].avatar = ev.target.result;
    persist(); render();
  };
  reader.readAsDataURL(file);
}
